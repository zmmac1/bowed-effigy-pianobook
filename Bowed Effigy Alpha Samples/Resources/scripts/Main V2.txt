on note


{if ($note_counter = 0)
    $ani_frame := 0
end if}

inc($note_counter)

    if ($is_ani_active = 0)

    $is_ani_active := 1
    $ani_frame := 0

    while ($note_counter > 0)
            if ($ani_frame < 54)
                set_control_par($lbl_ani_id, $CONTROL_PAR_PICTURE_STATE, $ani_frame)
                wait($frame_rate)
                inc($ani_frame)
                else
                $ani_frame := (($loop_start + 1) mod 237) + 54
                set_control_par($lbl_ani_id, $CONTROL_PAR_PICTURE_STATE, $ani_frame)
                wait($frame_rate)
                inc($loop_start)
            end if
        
    end while

    $is_ani_active := 0

    end if

end on

on release
    wait(200000)
    dec($note_counter)
    if ($note_counter = 0)
        $loop_start := 0
        $ani_frame := 300
        while ($ani_frame <= $MAX_FRAMES)
            $is_ani_active := 1
            inc($ani_frame)
            wait($frame_rate/3)
            set_control_par($lbl_ani_id, $CONTROL_PAR_PICTURE_STATE, $ani_frame)
        end while
        $is_ani_active := 0
    end if
end on
